   266	        """Record the intention on disk BEFORE the order leaves: a crash or a lost answer cannot cause a second order."""
   267

   269	        self.state["intents"][key] = {"id": key, "symbol": symbol, "side": side, "lots": lots, "sl": sl, "entry_ts": entry_ts,
   270	                                      "status": "SOUMIS", "at": self.clock().isoformat()}
   271	        for old in sorted(self.state["intents"])[:-INTENTS_KEPT]:
   272	            self.state["intents"].pop(old, None)
   273	        self.save()
   274	        return key

   283	    def resolve_intents(self, positions: list[Any]) -> list[dict[str, Any]]:
   284	        """Settle every intention left open (lost answer, crash, restart) from what the account really holds."""
   285	
   286	        settled = []
   287	        for intent in list(self.state["intents"].values()):
   288	            if intent["status"] not in PENDING:
   289	                continue
   290	            tag = f"{COMMENT}:{intent['id']}"
